function SectionText({ eyebrow, title, body }: { eyebrow?: string; title: string; body?: string }) {
  return (
    <motion.div
      initial="hidden"
      whileInView="visible"
      viewport={{ once: false, amount: 0.5 }}
      variants={fadeUp}
      className="max-w-2xl"

      </h2>
      {body && <p className="text-base md:text-lg text-[var(--text-secondary)] mt-4">{body}</p>}
    </motion.div>
  );
}

function GlassCard({ label, value }: { label: string; value: string }) {

      variants={fadeUp}

    </motion.div>
  );
}

      </div>

}
